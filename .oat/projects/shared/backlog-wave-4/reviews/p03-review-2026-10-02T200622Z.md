```bash
TSX_TSCONFIG_PATH=packages/cli/tsconfig.json node --import tsx --input-type=module --eval '
import assert from "node:assert/strict";
import { mkdir, mkdtemp, writeFile, readFile } from "node:fs/promises";
import { tmpdir } from "node:os";
import { join } from "node:path";
import { computeSyncPlan } from "@engine/compute-plan";
import { executeSyncPlan } from "@engine/execute-plan";
import { scanCanonical } from "@engine/scanner";
import { createTestAdapter } from "@engine/test-helpers";
import { loadManifest, saveManifest } from "@manifest/manager";
import { DEFAULT_SYNC_CONFIG } from "@config/sync-config";
import { detectDrift } from "@drift/detector";
async function main() {
  const root = await mkdtemp(join(tmpdir(), "bw4-p03-path-probe-"));
  const skill = join(root, ".agents", "skills", "skill-one");
  const manifestPath = join(root, ".oat", "sync", "manifest.json");
  await mkdir(skill, { recursive: true });
  await writeFile(join(skill, "SKILL.md"), "# skill\n");
  const adapter = createTestAdapter({ defaultStrategy: "copy" });
  const planFor = async manifest => computeSyncPlan({
    canonical: await scanCanonical(root, "project"), adapters: [adapter],
    manifest, scope: "project", scopeRoot: root,
    config: { ...DEFAULT_SYNC_CONFIG, defaultStrategy: "copy" }
  });
  let manifest = await loadManifest(manifestPath);
  await executeSyncPlan(await planFor(manifest), manifest, manifestPath);
  const synced = await loadManifest(manifestPath);
  const staleHash = "a".repeat(64);
  const expectFixed = process.env.P03_EXPECT_FIXED === "1";
  for (const prefix of ["", "./"]) {
    await saveManifest(manifestPath, {
      ...synced, entries: synced.entries.map(entry => ({
        ...entry, providerPath: prefix + entry.providerPath, contentHash: staleHash
      }))
    });
    manifest = await loadManifest(manifestPath);
    const plan = await planFor(manifest);
    assert.equal(plan.entries[0].operation, "skip");
    assert.ok(plan.entries[0].restampContentHash);
    await executeSyncPlan(plan, manifest, manifestPath);
    const after = await loadManifest(manifestPath);
    const next = await planFor(after);
    const drift = await detectDrift(after.entries[0], root);
    const outcome = {
      providerPath: after.entries[0].providerPath,
      plannedRestamp: true,
      hashRepaired: after.entries[0].contentHash !== staleHash,
      nextRestamp: next.entries[0].restampContentHash !== undefined,
      drift: drift.state
    };
    console.log(JSON.stringify(outcome));
    assert.equal(outcome.hashRepaired, prefix === "" || expectFixed);
    assert.equal(outcome.nextRestamp, prefix === "./" && !expectFixed);
  }
  console.log("fixture=" + root);
}
main().catch(error => { console.error(error); process.exitCode = 1; });
'
```
